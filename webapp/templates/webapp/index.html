interface State {
    clics: number;
}

const state: State = { clics: 0 };
const cta = document.getElementById("cta")!;
const out = document.getElementById("out")!;

cta.addEventListener("click", () => {
    state.clics += 1;
    out.hidden = false;
    out.textContent = `Clics : ${state.clics}`;
});
